type royalAuditedGame interface {
	Reset()
	GetChips() int
	SetChips(int)
}

func royalCanonicalSnapshot(data []byte, ignoreLog bool) []byte {
	var value map[string]interface{}
	if err := json.Unmarshal(data, &value); err != nil {
		panic(err)
	}
	if ignoreLog {
		delete(value, "al")
	}
	var normalize func(interface{}) interface{}
	normalize = func(v interface{}) interface{} {
		if v == nil {
			return []interface{}{}
		}
		switch x := v.(type) {
		case map[string]interface{}:
			for k, item := range x {
				x[k] = normalize(item)
			}
		case []interface{}:
			for i, item := range x {
				x[i] = normalize(item)
			}
		}
		return v
	}
	data, _ = json.Marshal(normalize(value))
	return data
}

func TestRoyalAuditCardRecoveryAndRefill(t *testing.T) {
	cases := []struct {
		name    string
		newGame func() royalAuditedGame
		start   func(royalAuditedGame) error
		finish  func(royalAuditedGame) error
	}{
		{"ThreeCard", func() royalAuditedGame { return NewDefaultThreeCard() }, func(g royalAuditedGame) error { return g.(*ThreeCard).Bet(10, 0) }, func(g royalAuditedGame) error { return g.(*ThreeCard).Play() }},
		{"CaribbeanStud", func() royalAuditedGame { return NewDefaultCaribbeanStud() }, func(g royalAuditedGame) error { return g.(*CaribbeanStud).Bet(10, 0) }, func(g royalAuditedGame) error { return g.(*CaribbeanStud).Play() }},
		{"LetItRide", func() royalAuditedGame { return NewDefaultLetItRide() }, func(g royalAuditedGame) error { return g.(*LetItRide).Bet(10) }, func(g royalAuditedGame) error {
			v := g.(*LetItRide)
			if err := v.LetItRideAction(); err != nil {
				return err
			}
			return v.LetItRideAction()
		}},
		{"UltimateTexasHoldem", func() royalAuditedGame { return NewDefaultUltimateTexasHoldem() }, func(g royalAuditedGame) error { return g.(*UltimateTexasHoldem).Bet(10, 0) }, func(g royalAuditedGame) error { return g.(*UltimateTexasHoldem).Play(4) }},
		{"CasinoWar", func() royalAuditedGame { return NewDefaultCasinoWar() }, func(g royalAuditedGame) error { return g.(*CasinoWar).Bet(10) }, func(g royalAuditedGame) error {
			v := g.(*CasinoWar)
			v.ResolveInitial()
			if v.GetPhase() == CasinoWarPhaseTieDecision {
				return v.GoToWar()
			}
			return nil
		}},
		{"RedDog", func() royalAuditedGame { return NewDefaultRedDog() }, func(g royalAuditedGame) error { return g.(*RedDog).Bet(10) }, func(g royalAuditedGame) error {
			v := g.(*RedDog)
			v.ResolveInitial()
			if v.GetPhase() == RedDogPhaseSpreadDecision {
				return v.Stay()
			}
			return nil
		}},
		{"PaiGow", func() royalAuditedGame { return NewDefaultPaiGow() }, func(g royalAuditedGame) error { return g.(*PaiGow).Bet(10) }, func(g royalAuditedGame) error {
			v := g.(*PaiGow)
			h := v.GetHint()
			if h == nil {
				t.Fatal("missing legal split")
			}
			return v.SetHands(h.LowIdx0, h.LowIdx1)
		}},
		{"DragonTiger", func() royalAuditedGame { return NewDefaultDragonTiger() }, func(g royalAuditedGame) error { return g.(*DragonTiger).Bet(10, DragonTigerBetDragon) }, func(g royalAuditedGame) error { return nil }},
		{"DeucesWild", func() royalAuditedGame { return NewDeucesWildVideoPoker() }, func(g royalAuditedGame) error { return g.(*VideoPoker).Bet(5) }, func(g royalAuditedGame) error { return g.(*VideoPoker).Hold([]int{0, 2}) }},
	}
	for _, c := range cases {
		t.Run(c.name, func(t *testing.T) {
			for i := 0; i < 25; i++ {
				g := c.newGame()
				if err := c.start(g); err != nil {
					t.Fatal(err)
				}
				before, err := json.Marshal(g)
				if err != nil {
					t.Fatal(err)
				}
				restored := reflect.New(reflect.TypeOf(g).Elem()).Interface().(royalAuditedGame)
				if err = json.Unmarshal(before, restored); err != nil {
					t.Fatal(err)
				}
				snapshot, err := json.Marshal(restored)
				if err != nil || !bytes.Equal(royalCanonicalSnapshot(before, false), royalCanonicalSnapshot(snapshot, false)) {
					t.Fatalf("state changed on reload: %v", err)
				}
				if err = c.finish(g); err != nil {
					t.Fatal(err)
				}
				if err = c.finish(restored); err != nil {
					t.Fatal(err)
				}
				a, _ := json.Marshal(g)
				b, _ := json.Marshal(restored)
				if !bytes.Equal(royalCanonicalSnapshot(a, true), royalCanonicalSnapshot(b, true)) {
					t.Fatalf("same action after recovery produced a different game result: %s %s", a, b)
				}
				if !bytes.Equal(royalCanonicalSnapshot(a, false), royalCanonicalSnapshot(b, false)) {
					t.Log("action history differs after recovery despite equal economic result")
				}
			}
			g := c.newGame()
			g.SetChips(0)
			g.Reset()
			if g.GetChips() <= 0 {
				t.Fatal("expected upstream practice refill")
			}
			t.Logf("confirmed automatic refill to %d practice chips", g.GetChips())
		})
	}
}

func TestRoyalAuditThreeCardExactPairPlus(t *testing.T) {
	deck := NewTrumpCards(0).deck
	counts := map[int]int{}
	total, payout := 0, 0
	for a := 0; a < 50; a++ {
		for b := a + 1; b < 51; b++ {
			for c := b + 1; c < 52; c++ {
				rank := evalThreeCardHand([]*Card{deck[a], deck[b], deck[c]})
				counts[rank]++
				g := ThreeCard{pairPlusBet: 10, playerHandRank: rank}
				g.evaluatePairPlus()
				payout += g.pairPlusPayout
				total++
			}
		}
	}
	if total != 22100 {
		t.Fatal(total)
	}
	expected := map[int]int{ThreeCardHandStraightFlush: 48, ThreeCardHandThreeOfAKind: 52, ThreeCardHandStraight: 720, ThreeCardHandFlush: 1096, ThreeCardHandPair: 3744, ThreeCardHandHighCard: 16440}
	if !reflect.DeepEqual(counts, expected) {
		t.Fatalf("wrong hand counts: %v", counts)
	}
	t.Logf("Pair Plus exact return: payout=%d stake=%d RTP=%.12f%%", payout, total*10, float64(payout)/float64(total*10)*100)
}

func TestRoyalAuditDragonTigerExact(t *testing.T) {
	deck := NewTrumpCards(0).deck
	for _, kind := range []int{DragonTigerBetDragon, DragonTigerBetTiger, DragonTigerBetTie} {
		payout, total := 0, 0
		for a := 0; a < 52; a++ {
			for b := 0; b < 52; b++ {
				if a == b {
					continue
				}
				g := DragonTiger{dragonCard: deck[a], tigerCard: deck[b], betType: kind, betAmount: 10}
				g.judge()
				payout += g.payout
				total++
			}
		}
		expected := 25740
		if kind == DragonTigerBetTie {
			expected = 14040
		}
		if payout != expected || total != 2652 {
			t.Fatalf("wrong return %d %d", payout, total)
		}
		t.Logf("bet type %d exact RTP=%.12f%% (%d/%d)", kind, float64(payout)/float64(total*10)*100, payout, total*10)
	}
}

func TestRoyalAuditPaiGowCommissionRounding(t *testing.T) {
	g := NewDefaultPaiGow()
	g.playerHighHand = []*Card{NewCard(CardDesignSpade, 1, false), NewCard(CardDesignSpade, 13, false), NewCard(CardDesignSpade, 12, false), NewCard(CardDesignSpade, 11, false), NewCard(CardDesignSpade, 10, false)}
	g.playerLowHand = []*Card{NewCard(CardDesignHeart, 13, false), NewCard(CardDesignDiamond, 13, false)}
	g.dealerHighHand = []*Card{NewCard(CardDesignHeart, 9, false), NewCard(CardDesignDiamond, 9, false), NewCard(CardDesignClover, 8, false), NewCard(CardDesignHeart, 6, false), NewCard(CardDesignDiamond, 4, false)}
	g.dealerLowHand = []*Card{NewCard(CardDesignHeart, 3, false), NewCard(CardDesignDiamond, 2, false)}
	g.bet = 10
	g.resolve()
	if g.result != GameResultWin || g.commission != 0 || g.payout != 20 {
		t.Fatalf("unexpected rounding fixture: %d %d", g.commission, g.payout)
	}
	t.Log("5 percent commission on 10 upstream chips rounds to zero; adapter needs monetary minor units")
}

func TestRoyalAuditThreeCardFoldSideBet(t *testing.T) {
	g := NewDefaultThreeCard()
	if err := g.Bet(10, 10); err != nil {
		t.Fatal(err)
	}
	g.playerHand = []*Card{NewCard(CardDesignSpade, 5, false), NewCard(CardDesignSpade, 6, false), NewCard(CardDesignSpade, 7, false)}
	before := g.GetChips()
	if err := g.Fold(); err != nil {
		t.Fatal(err)
	}
	if g.GetPairPlusPayout() != 410 || g.GetChips() != before {
		t.Fatalf("unexpected fixture: payout=%d balance=%d", g.GetPairPlusPayout(), g.GetChips())
	}
	t.Logf("Fold reports winning Pair Plus payout of %d, but balance stays %d", g.GetPairPlusPayout(), g.GetChips())
}

func TestRoyalAuditCaribbeanJackpotExact(t *testing.T) {
	deck := NewTrumpCards(0).deck
	var payout, total int64
	for a := 0; a < 48; a++ {
		for b := a + 1; b < 49; b++ {
			for c := b + 1; c < 50; c++ {
				for d := c + 1; d < 51; d++ {
					for e := d + 1; e < 52; e++ {
						rank := evalFiveCardHand([]*Card{deck[a], deck[b], deck[c], deck[d], deck[e]})
						g := CaribbeanStud{jackpotBet: 10, playerHandRank: rank}
						g.evaluateJackpot()
						payout += int64(g.jackpotPayout)
						total++
					}
				}
			}
		}
	}
	if total != 2598960 || payout != 12018000 {
		t.Fatalf("wrong exact count: %d %d", total, payout)
	}
	t.Logf("fixed jackpot side bet exact RTP=%.12f%% (%d/%d), no progressive pot", float64(payout)/float64(total*10)*100, payout, total*10)
}

func TestRoyalAuditDeucesWildMutableDealHistory(t *testing.T) {
	g := NewDeucesWildVideoPoker()
	if err := g.Bet(5); err != nil {
		t.Fatal(err)
	}
	before, _ := json.Marshal(g)
	var restored VideoPoker
	if err := json.Unmarshal(before, &restored); err != nil {
		t.Fatal(err)
	}
	initial, _ := json.Marshal(g.actionLog[1].Cards)
	if err := g.Hold([]int{0, 2}); err != nil {
		t.Fatal(err)
	}
	if err := restored.Hold([]int{0, 2}); err != nil {
		t.Fatal(err)
	}
	originalLog, _ := json.Marshal(g.actionLog[1].Cards)
	restoredLog, _ := json.Marshal(restored.actionLog[1].Cards)
	if bytes.Equal(initial, originalLog) || !bytes.Equal(initial, restoredLog) {
		t.Fatal("unexpected action-log alias fixture")
	}
	t.Log("draw mutates original deal history; JSON recovery breaks the alias and keeps the initial hand")
}
